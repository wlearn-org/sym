cmake_minimum_required(VERSION 3.16)
project(sym C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

set(CORE_SOURCES src/sym.c src/sym_tree_refine.c src/sym_tree_io.c src/sym_family.c src/sym_family_lm.c src/sym_family_io.c src/wl_api.c)

add_library(sym SHARED ${CORE_SOURCES})
target_include_directories(sym PUBLIC src)
target_compile_definitions(sym PRIVATE _POSIX_C_SOURCE=200809L)
target_compile_options(sym PRIVATE -Wall -Wextra -Wpedantic -O2)
target_link_libraries(sym m)
set_target_properties(sym PROPERTIES OUTPUT_NAME sym)

if(BUILD_TESTING)
    enable_testing()
    add_executable(test_sym test/test_c.c)
    target_include_directories(test_sym PRIVATE src)
    target_link_libraries(test_sym sym m)
    add_test(NAME sym_contract COMMAND test_sym)
    add_executable(test_search test/test_search.c)
    target_include_directories(test_search PRIVATE src)
    target_link_libraries(test_search sym m)
    add_test(NAME sym_search COMMAND test_search)
    add_executable(test_family test/test_family.c)
    target_include_directories(test_family PRIVATE src)
    target_link_libraries(test_family sym m)
    add_test(NAME sym_family COMMAND test_family)
    add_executable(test_family_search test/test_family_search.c)
    target_include_directories(test_family_search PRIVATE src)
    target_link_libraries(test_family_search sym m)
    add_test(NAME sym_family_search COMMAND test_family_search)
endif()
